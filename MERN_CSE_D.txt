Aggregate functions

Syntax:
db.collection_name.aggregate([{agg fun1},{agg fun2},{agg fun3}])


$match => used to filter the records

db.products.aggregate([
  {$match:{"category":"Laptop"}}
]).forEach(doc=>print(JSON.stringify(doc)))


$match and $sort

db.products.aggregate([
  {$match:{"category":"Laptop"}},
  {$sort:{"price":-1}}
]).forEach(doc=>print(JSON.stringify(doc)))


$match and $sort and $limit(display only limited records)

$project => used to display particular colunms


db.products.aggregate([
  {$match:{"category":"Laptop"}},
  {$sort:{"price":-1}},
  {$limit:2},
  {$project:{_id:0,name:1,brand:1,price:1}}
]).forEach(doc=>print(JSON.stringify(doc)))


$group=> it is used to group the documents

$sum => used to find the total of the particular

$min => used to print minimum value in particular col
$max => used to print maximum value in particular col
$avg => used to print average value in particular col

db.products.aggregate([
  {$group:{
    _id:"$category",
    "totalproducts":{$sum:1},
    "lowestprice":{$min:"$price"}}}

$lookup
   -from (which table need to connect) eg:sections
   -localField (local ref field to connect with another table (sec_id)
   -foreignField (in another table what is the ref key (sec_id)
   -as (used to store fetch data from another table in one key
        eg: as:"Section_details"



db.students.aggregate([
  {$lookup:{
    from:"sections",
    localField:"sec_id",
    foreignField:"sec_id",
    as:"section_details"
  }}
]).forEach(doc=>print(JSON.stringify(doc)))


#multiple tables lookup query
db.students.aggregate([
  {$lookup:{from:"sections",localField:"sec_id",foreignField:"sec_id",
   as:"sec_details"}},
  {$lookup:{from:"departments",localField:"dept_id",foreignField:"dept_id",
    as:"dept_details"}}  
]).forEach(doc=>print(JSON.stringify(doc)))
